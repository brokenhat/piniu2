@app.route('/edit', methods=['GET', 'POST'])
@login_required  # decorator
def edit():
    form = PostForm()
    pageform = NewPage()

    page = request.args.get('page', 1, type=int)
    posts = Post.query.order_by(Post.timestamp.desc()).paginate(
        page, app.config['POSTS_PER_PAGE'], False)
    next_url = url_for('edit', page=posts.next_num) \
        if posts.has_next else None
    prev_url = url_for('edit', page=posts.next_num) \
        if posts.has_prev else None

# Adding pages
    pages = Page.query.order_by(Page.id.desc())
    if pageform.validate_on_submit():
        page = Page(title=pageform.title.data)
        db.session.add(page)
        db.session.commit()
        flash('Strona została dodana.')
        return redirect(url_for('edit'))

# Adding posts
    if form.validate_on_submit():
        post = Post(body=form.post.data, author=current_user)
        db.session.add(post)
        db.session.commit()
        flash('Zmiany zostały opublikowane!')
        return redirect(url_for('edit'))

    return render_template('edit.html', title='Strona główna - Edycja',
                            form=form, posts=posts.items,
                            next_url=next_url,
                            prev_url=prev_url,
                            pages=pages, pageform=pageform)
